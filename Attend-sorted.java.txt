package attendance;

import java.io.IOException;
import java.io.PrintWriter;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import javax.servlet.ServletException;
import javax.servlet.annotation.WebServlet;
import javax.servlet.http.HttpServlet;
import javax.servlet.http.HttpServletRequest;
import javax.servlet.http.HttpServletResponse;

/**
 * Optional variant for Q3: the Attendee List is sorted by Student Id.
 *
 * Only two things differ from the lab's Attend.java:
 *   1. the "Attendee List" branch copies sid2seat.entrySet() into a List and sorts it;
 *   2. a compareStudentId() helper compares numerically so "100" sorts after "99".
 *
 * To use it: replace src/java/attendance/Attend.java with this file and redeploy.
 */
@WebServlet("/Attend")
public class Attend extends HttpServlet {

    private static final long serialVersionUID = 1L;

    private static final Map<String, String> sid2seat = new HashMap<String, String>();

    @Override
    protected void doGet(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        handle(request, response);
    }

    @Override
    protected void doPost(HttpServletRequest request, HttpServletResponse response)
            throws ServletException, IOException {
        handle(request, response);
    }

    private void handle(HttpServletRequest request, HttpServletResponse response)
            throws IOException {
        response.setContentType("text/html;charset=UTF-8");
        PrintWriter out = response.getWriter();

        String studentId = request.getParameter("studentId");
        String seatId    = request.getParameter("seatId");

        out.println("Simple Attendance by Seat Id<br><br>");

        if (studentId == null) {
            // ----- Attendee List: dump the map, sorted by Student Id -----
            synchronized (sid2seat) {
                if (sid2seat.isEmpty()) {
                    out.println("No sign-in record yet.<br><br>");
                } else {
                    // Q3 change 1: HashMap has no order, so copy and sort the entries.
                    List<Map.Entry<String, String>> rows =
                            new ArrayList<Map.Entry<String, String>>(sid2seat.entrySet());
                    rows.sort(new Comparator<Map.Entry<String, String>>() {
                        @Override
                        public int compare(Map.Entry<String, String> a,
                                           Map.Entry<String, String> b) {
                            return compareStudentId(a.getKey(), b.getKey());
                        }
                    });
                    for (Map.Entry<String, String> e : rows) {
                        out.println(e.getKey() + " : " + e.getValue() + "<br>");
                    }
                }
            }
            out.println("<br><a href=\"index.html\">Attendance Submission</a>");
        } else if (!studentId.trim().isEmpty()) {
            // ----- Record the pair -----
            synchronized (sid2seat) {
                sid2seat.put(studentId.trim(), seatId == null ? "" : seatId.trim());
            }
            out.println(studentId + ":" + seatId + " is recorded:<br><br>");
            out.println("<a href=\"index.html\">Attendance Submission</a> | "
                      + "<a href=\"Attend\">Attendee List</a>");
        } else {
            // ----- Blank student id -----
            out.println("Student Id can not be blank<br><br>");
            out.println("<a href=\"index.html\">Attendance Submission</a> | "
                      + "<a href=\"Attend\">Attendee List</a>");
        }
    }

    /**
     * Q3 change 2: Student Ids are Strings, so a plain lexicographic compare puts
     * "100" before "99". Compare numerically when both sides are numbers, and fall
     * back to a case-insensitive string compare otherwise.
     */
    private static int compareStudentId(String a, String b) {
        try {
            return Long.compare(Long.parseLong(a.trim()), Long.parseLong(b.trim()));
        } catch (NumberFormatException notNumeric) {
            return a.compareToIgnoreCase(b);
        }
    }
}
